/* ---------------- PARTICLES (OPTIMIZED) ---------------- */ const canvas = document.getElementById('particles'); const ctx = canvas.getContext('2d'); function resize() { canvas.width = window.innerWidth; canvas.height = window.innerHeight; } resize(); window.addEventListener("resize", resize); /* MOBILE PERFORMANCE BOOST */ const PARTICLE_COUNT = window.innerWidth < 768 ? 40 : 100; let particles = []; for (let i = 0; i < PARTICLE_COUNT; i++) { particles.push({ x: Math.random() * canvas.width, y: Math.random() * canvas.height, r: Math.random() * 1.5, d: Math.random() * 1 + 0.2 }); } /* FPS LIMITER (huge mobile improvement) */ let lastTime = 0; function animateParticles(time = 0) { if (time - lastTime > 33) { ctx.clearRect(0, 0, canvas.width, canvas.height); ctx.fillStyle = 'rgba(255,255,255,0.4)'; particles.forEach(p => { ctx.beginPath(); ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2); ctx.fill(); p.y += p.d; if (p.y > canvas.height) { p.y = 0; p.x = Math.random() * canvas.width; } }); lastTime = time; } requestAnimationFrame(animateParticles); } animateParticles(); /* ---------------- CURSOR (DISABLED ON MOBILE) ---------------- */ if (window.innerWidth > 768) { const glow = document.getElementById('cursorGlow'); window.addEventListener('mousemove', e => { requestAnimationFrame(() => { glow.style.left = e.clientX + 'px'; glow.style.top = e.clientY + 'px'; }); }); }
